# [229. Majority Element II](https://leetcode.com/problems/majority-element-ii/)
```py
class Solution(object):
    def majorityElement(self, nums):
        freq = {}
        res = []
        for i in nums:
            freq[i] = freq.get(i, 0) + 1
        for i,j in freq.items():
            if j > len(nums) / 3:
                res.append(i)
        return res

# [178. Longest subarray with sum K](https://takeuforward.org/practice/dsa/longest-subarray-with-sum-k)
```py
arr = [1,2,3,1,1,1,1,3,3]
target = 6
currsum = arr[0]
i = j = res = 0
while j < len(arr):
    while currsum > target and i <= j:
        currsum -= arr[i]
        i += 1
    if currsum == target:
        res = max(res, j - i + 1)
    j += 1
    if j < len(arr):
        currsum += arr[j]
print(res)

class Solution(object):
    def searchInsert(self, nums, target):
        # nums.append(target)
        # nums.sort()
        # for i in range(len(nums)):
        #     if nums[i] == target:
        #         return i
        low = 0
        high = len(nums) - 1

        while low <= high:
            mid = (low + high) // 2
            if nums[mid] >= target:
                high = mid - 1
            else:
                low = mid + 1
        return low

# [76. Minimum Window Substring](https://leetcode.com/problems/minimum-window-substring/description/)
```py
class Solution(object):
    def minWindow(self, s, t):

        # mapi stores how many characters of t are still needed
        mapi = [0] * 256

        for i in range(len(t)):
            mapi[ord(t[i])] += 1

        l = r = 0
        start = -1
        count = 0
        minlen = float('inf')

        while r < len(s):

            # If this character is still needed
            if mapi[ord(s[r])] > 0:
                count += 1

            # Decrease the required count
            mapi[ord(s[r])] -= 1

            # We have all characters required by t
            while count == len(t):

                # Check if this is the smallest window
                if (r - l + 1) < minlen:
                    minlen = r - l + 1
                    start = l

                # Remove the left character
                mapi[ord(s[l])] += 1

                # If it is needed again,
                # our window is no longer valid
                if mapi[ord(s[l])] > 0:
                    count -= 1

                # Move left pointer
                l += 1

            r += 1

        if start == -1:
            return ""

        return s[start:start + minlen]

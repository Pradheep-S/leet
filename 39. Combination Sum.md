# [39. Combination Sum](https://leetcode.com/problems/combination-sum/description/)
```py
class Solution(object):
    def combinationSum(self, candidates, target):

        def recur(ind, arr, target, res, ds):

            # Reached the end
            if ind == len(arr):
                if target == 0:
                    res.append(ds[:])
                return

            # Take current element
            if arr[ind] <= target:
                ds.append(arr[ind])

                # Stay at same index because
                # we can use the same number again
                recur(ind, arr, target - arr[ind], res, ds)

                # Remove the element before backtracking
                ds.pop()

            # Don't take current element
            recur(ind + 1, arr, target, res, ds)

        res = []
        ds = []
        recur(0, candidates, target, res, ds)

        return res

# [45. Jump Game II](https://leetcode.com/problems/jump-game-ii/description/)
```py
class Solution(object):
    def jump(self, nums):
        jumps = 0
        l = r = 0
        while r < len(nums) - 1:
            farthest = 0
            # to find the most distance the r can move
            for i in range(l, r + 1):
                farthest = max(farthest, i + nums[i])
            l = r + 1
            r = farthest
            jumps += 1 # range has took a jump
        return jumps

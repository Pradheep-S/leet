# [1423. Maximum Points You Can Obtain from Cards](https://leetcode.com/problems/maximum-points-you-can-obtain-from-cards/description/)
```py
class Solution(object):
    def maxScore(self, cardPoints, k):
        lsum, rsum = 0, 0
        maxsum = 0
        for i in range(k):
            lsum += cardPoints[i]
        maxsum = lsum
        r = len(cardPoints) - 1 # right pointer to start adding from last when we reduce the front 
        for i in range(k-1, -1, -1):
            lsum -= cardPoints[i]
            rsum += cardPoints[r]
            r -= 1
            maxsum = max(maxsum, lsum + rsum)
        return maxsum

# [424. Longest Repeating Character Replacement](https://leetcode.com/problems/longest-repeating-character-replacement/description/)

class Solution(object):
    def characterReplacement(self, s, k):
        l = r = 0
        mapi = [0]*26
        maxf = maxlen = 0
        while r < len(s):
            mapi[ord(s[r]) - ord('A')] += 1 
            maxf = max(maxf, mapi[ord(s[r]) - ord('A')]) # gives us the most repeated letter in our substr

            if((r - l + 1) - maxf) > k: # if swaps req. greater than least el.
            # while ((r - l + 1) - maxf) > k:
                mapi[ord(s[l]) - ord('A')] -= 1
                # tempmax = max(mapi)
                # maxf = max(maxf, tempmax)
                l += 1
            if ((r - l + 1) - maxf <= k): # valid substr
                maxlen = max(maxlen, r - l + 1)
            r += 1
            
        return maxlen

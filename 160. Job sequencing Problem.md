# [160. Job sequencing Problem](https://takeuforward.org/practice/dsa/job-sequencing-problem)

class Solution(object):
    def JobScheduling(self, Jobs):

        # Sort jobs by profit in descending order
        Jobs.sort(key=lambda x: x[2], reverse=True)

        # Find the maximum deadline
        max_deadline = 0
        for job in Jobs:
            max_deadline = max(max_deadline, job[1])

        # Create slots
        # slot[i] = job scheduled at time i
        slot = [-1] * (max_deadline + 1)

        count = 0
        profit = 0

        # Try each job
        for job in Jobs:
            job_id = job[0]
            deadline = job[1]
            job_profit = job[2]

            # Try to put the job in the latest possible slot
            for j in range(deadline, 0, -1):

                # If this slot is empty
                if slot[j] == -1:
                    slot[j] = job_id
                    count += 1
                    profit += job_profit
                    break

        return [count, profit]

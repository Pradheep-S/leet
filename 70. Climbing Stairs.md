class Solution(object):
    def climbStairs(self, n):
        # # recursive code
        # if n <= 1:
        #     return 1
        # left = self.climbStairs(n-1)
        # right = self.climbStairs(n-2)
        # return left + right
        dp = [0] * (n+1)
        dp[0] = 1
        dp[1] = 1
        for i in range(2, n+1):
            dp[i] = dp[i-1] + dp[i-2]
        return dp[n]
        # this is called memoization

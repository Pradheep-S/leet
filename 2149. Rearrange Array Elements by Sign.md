# [2149. Rearrange Array Elements by Sign](https://leetcode.com/problems/rearrange-array-elements-by-sign/description/)
```py
class Solution(object):
    def rearrangeArray(self, nums):
        parray = []
        narray = []
        res = []
        for i in nums:
            if i > 0:
                parray.append(i)
            else:
                narray.append(i)
        i = j = 0
        while i < len(parray) and j < len(narray):
            res.append(parray[i])
            res.append(narray[j])
            i+=1
            j+=1
        return res

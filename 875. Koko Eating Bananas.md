# [875. Koko Eating Bananas](https://leetcode.com/problems/koko-eating-bananas/description/)

class Solution(object):
    def minEatingSpeed(self, piles, h):
        low = 1
        high = max(piles)
        speed = low
        while low <= high:
            mid = (low + high) // 2 #mid is the speed here
            totaltime = 0
            for i in piles:
                totaltime += (i + mid - 1) // mid #ceil division
            if totaltime <= h:
                speed = mid
                high = mid - 1
            else:
                low = mid + 1
        return speed

class Solution(object):
    def setZeroes(self, matrix):
        ##separate marker arrays
        # m = len(matrix)
        # n = len(matrix[0])
        # col = [0] * n
        # row = [0] * m
        # for i in range(m):
        #     for j in range(n):
        #         if matrix[i][j] == 0:
        #             col[j] = 1
        #             row[i] = 1
        # for i in range(m):
        #     for j in range(n):
        #         if col[j] == 1 or row[i] == 1:
        #             matrix[i][j] = 0
        m = len(matrix)
        n = len(matrix[0])
        col0 = 1

        # Mark rows and columns
        for i in range(m):
            if matrix[i][0] == 0:
                col0 = 0

            for j in range(1, n):
                if matrix[i][j] == 0:
                    matrix[i][0] = 0
                    matrix[0][j] = 0

        # Set zeroes using the markers
        for i in range(m - 1, -1, -1):
            for j in range(n - 1, 0, -1):
                if matrix[i][0] == 0 or matrix[0][j] == 0:
                    matrix[i][j] = 0

            if col0 == 0:
                matrix[i][0] = 0

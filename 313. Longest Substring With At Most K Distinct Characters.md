# [313. Longest Substring With At Most K Distinct Characters](https://takeuforward.org/practice/dsa/longest-substring-with-at-most-k-distinct-characters)
```py
class Solution:
    def kDistinctChar(self, s, k):

        l = r = 0
        maxlen = 0
        seen = {}

        while r < len(s):

            # Add the current character
            seen[s[r]] = seen.get(s[r], 0) + 1

            # If we have more than k distinct characters
            if len(seen) > k:

                # Character leaving the window
                left_el = s[l]

                # Decrease its frequency
                seen[left_el] -= 1

                # Move left pointer
                l += 1

                # Remove character if its frequency becomes 0
                if seen[left_el] == 0:
                    del seen[left_el]

            # Update maximum length
            if len(seen) <= k:
                maxlen = max(maxlen, r - l + 1)

            r += 1

        return maxlen

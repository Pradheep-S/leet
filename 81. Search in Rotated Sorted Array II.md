# [81. Search in Rotated Sorted Array II](https://leetcode.com/problems/search-in-rotated-sorted-array-ii/description/)
```py
class Solution(object):
    def search(self, nums, target):
        low = 0
        high = len(nums) - 1
        while low <= high:
            mid = (low + high) // 2
            if nums[mid] == target:
                return True
            # to shrink nums, so that we reduce dupes(then only we can find the sorted part)
            if nums[mid] == nums[low] and nums[mid] == nums[high]:
                low += 1
                high -= 1
                continue

            if nums[low] <= nums[mid]: # means left is sorted
                if nums[low] <= target and nums[mid] >= target:
                    high = mid - 1
                else:
                    low = mid + 1
            else: # this is when the right of mid is sorted
                if nums[mid] <= target and nums[high] >= target:
                    low = mid + 1
                else:
                    high = mid - 1
        return False

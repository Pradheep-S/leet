# [742. Second Highest Occurring Element](https://takeuforward.org/practice/dsa/second-highest-occurring-element)
## Solution
```py
def second_most_frequent(arr):
    freq = {}

    # Count frequency
    for x in arr:
        freq[x] = freq.get(x, 0) + 1

    # Find highest and second-highest frequency
    first = -1
    second = -1

    for x in freq:
        if freq[x] > first:
            second = first
            first = freq[x]
        elif freq[x] > second and freq[x] != first:
            second = freq[x]

    # No second frequency
    if second == -1:
        return -1

    # Find smallest element having second frequency
    ans = float('inf')

    for x in freq:
        if freq[x] == second:
            ans = min(ans, x)

    return ans

class Solution(object):
    def minDays(self, bloomDay, m, k):
        def nconsecutives(arr, m, k):
            if len(arr) < (m * k):
                return False
            bcount = count = 0
            for i in arr:
                if i == 1:
                    count += 1
                else:
                    count = 0
                    
                if count == k:
                    bcount += 1
                    count = 0
            return (bcount >= m) #made mistake here,  since atleast 
        n = len(bloomDay)
        low = min(bloomDay)
        high = max(bloomDay)
        res = -1
        while low <= high:
            mid = (low + high) // 2
            arr = [0] * n
            for i in range(n):
                if bloomDay[i] <= mid:
                    arr[i] = 1
            cons = nconsecutives(arr,m,k)
            if cons:
                res = mid
                high = mid - 1
            else:
                low = mid + 1
        return res

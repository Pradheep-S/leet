# [455. Assign Cookies](https://leetcode.com/problems/assign-cookies/description/)
```py
class Solution(object):
    def findContentChildren(self, g, s):
        i = j = 0
        g.sort()
        s.sort()
        count = 0
        while i < len(g) and j < len(s):
            if s[j] >= g[i]: # child satisfied with cookie size
                count += 1
                i += 1
                j += 1
            else: # child not satisfied with size, so move to the next cookie
                j += 1

        return count

# [1248. Count Number of Nice Subarrays](https://leetcode.com/problems/count-number-of-nice-subarrays/description/)
```py
class Solution(object):
    def numberOfSubarrays(self, nums, k):
        def lessequal(k):
            if k < 0:
                return 0
            l = r = 0
            total = 0
            count = 0
            while r < len(nums):
                total += (nums[r] % 2) # odd or even
                while total > k:
                    total -= (nums[l] % 2)
                    l += 1
                count += (r - l + 1)
                r += 1
            return count
        # same as the binary subarray with sum k problem
        return lessequal(k) - lessequal(k-1)

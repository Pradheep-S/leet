# [678. Valid Parenthesis String](https://leetcode.com/problems/valid-parenthesis-string/description/)
```py
class Solution(object):
    def checkValidString(self, s):
        # with the idea that range of being [-1,0,1]
        min = max = 0
        for i in s:
            if i == '(':
                min += 1
                max += 1
            elif i == ')':
                min -= 1
                max -= 1
            else:
                min -= 1
                max += 1
            # this is handlind min to maintain the range[0.1]
            if min < 0: 
                min = 0
            # for edge cases like starting with a ) 
            if max < 0:
                return False
        
        return (min == 0)

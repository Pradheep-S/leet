# [930. Binary Subarrays With Sum](https://leetcode.com/problems/binary-subarrays-with-sum/description/)
```py
class Solution(object):
    def numSubarraysWithSum(self, nums, goal):
        def lessorequal(goal):
            if goal < 0: # edge case to handle where goal = 0
                return 0
            l = r = 0
            sumtill = count = 0
            while r < len(nums):
                sumtill += nums[r]
                while sumtill > goal: # to move l until sum becomes less or equal
                    sumtill -= nums[l]
                    l += 1
                count += (r - l + 1) # add all the subarrs possible before that index
                r += 1
            return count
        # to get the subs equal to goal alone
        return lessorequal(goal) - lessorequal(goal - 1)

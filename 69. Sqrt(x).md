# [69. Sqrt(x)](https://leetcode.com/problems/sqrtx/description/)
```py
class Solution(object):
    def mySqrt(self, x):
        if x < 2:
            return x
        low = 1
        high = x
        ans = 0
        while low <= high:
            mid = (low + high) // 2
            # checking square of the mid and eliminating the remaining half
            if mid*mid <= x:
                ans = mid
                low = mid + 1
            else:
                high = mid - 1
        return ans

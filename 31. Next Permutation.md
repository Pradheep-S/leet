class Solution(object):
    def nextPermutation(self, nums):
        # 1. Find first nums[i] < nums[i+1] from right
        # 2. Find number > nums[i] from right
        # 3. Swap them
        # 4. Reverse the part after i
        ind = -1
        for i in range(len(nums)-2, -1, -1):
            if nums[i] < nums[i+1]:
                ind = i
                break
        #this is the last possible perm so reverse it to get the 1st
        if ind == -1:
            nums.reverse()
            return
        for i in range(len(nums)-1, ind, -1):
            if nums[i] > nums[ind]:
                nums[i], nums[ind] = nums[ind], nums[i] # to get the least possible next perm
                break
        nums[ind+1:] = reversed(nums[ind+1:])
        # break is important in this codes' loops

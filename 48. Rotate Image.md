# [48. Rotate Image](https://leetcode.com/problems/rotate-image/description/)

class Solution(object):
    def rotate(self, matrix):
        n = len(matrix)
        #transpose
        for i in range(n):
            for j in range(i+1, n):
                matrix[i][j], matrix[j][i] = matrix[j][i], matrix[i][j]
        #reverse each row
        for i in range(n):
            matrix[i].reverse()

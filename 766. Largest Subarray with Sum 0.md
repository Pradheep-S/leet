# [Largest Subarray with Sum 0](https://takeuforward.org/practice/dsa/largest-subarray-with-sum-0)
```py
arr = [15, -2, 2, -8, 1, 7, 10, 23]

hash = {0: -1}
prefixsum = 0
maxi = 0

for i, n in enumerate(arr):
    prefixsum += n

    if prefixsum in hash:

        # Length of the zero-sum subarray
        count = i - hash[prefixsum]

        maxi = max(maxi, count)

        print(f"Subarray found from index {hash[prefixsum] + 1} to {i}")

    else:
        # Store only the first occurrence
        hash[prefixsum] = i

print(maxi)

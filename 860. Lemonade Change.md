# [860. Lemonade Change](https://leetcode.com/problems/lemonade-change/description/)

class Solution(object):
    def lemonadeChange(self, bills):
        five = 0
        ten = 0 
        twenty = 0 # not actually required
        for i in bills:
            if i == 5:
                five += 1
            elif i == 10:
                if five:
                    five -= 1
                    ten += 1
                else:
                    return False
            else:
                if ten and five:
                    ten -= 1
                    five -= 1
                elif five >= 3:
                    five -= 3
                else:
                    return False
        return True

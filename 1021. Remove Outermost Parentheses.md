# [1021. Remove Outermost Parentheses](https://leetcode.com/problems/remove-outermost-parentheses/description/)
```py
class Solution(object):
    def removeOuterParentheses(self, s):
        res = ""
        depth = 0
        for i in s:
            if i == '(':
                if depth > 0: # means it is no the outermost
                    res += i
                depth += 1
            else:
                depth -= 1
                if depth > 0:
                    res += i
        return res

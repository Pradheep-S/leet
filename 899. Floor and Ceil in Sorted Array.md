# [899. Floor and Ceil in Sorted Array](https://takeuforward.org/practice/dsa/floor-and-ceil-in-sorted-array)

class Solution(object):
    def getFloorAndCeil(self, nums, x):

        low = 0
        high = len(nums) - 1

        floor = -1
        ceil = -1

        while low <= high:

            mid = (low + high) // 2

            if nums[mid] == x:
                # If x exists, it is both floor and ceil
                floor = nums[mid]
                ceil = nums[mid]
                return [floor, ceil]

            elif nums[mid] < x:
                # nums[mid] can be the floor
                floor = nums[mid]

                # Try to find a larger value
                low = mid + 1

            else:
                # nums[mid] can be the ceil
                ceil = nums[mid]

                # Try to find a smaller value
                high = mid - 1

        return [floor, ceil]

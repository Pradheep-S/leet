# [8. String to Integer (atoi)](https://leetcode.com/problems/string-to-integer-atoi/description/)
```py
class Solution(object):
    def myAtoi(self, s):
        s = s.strip()
        if not s: # empty string
            return 0
        sign = 1
        result = 0
        if s[0] == '-'or s[0] =='+':
            if s[0] == '-':
                sign = -1
            s = s[1:] # remove the sign
        for i in s:
            if not i.isdigit():
                break
            result = result * 10 + int(i)
        result = result * sign

        if result < -2**31: # given in question
            return -2**31
        if result > 2**31 - 1:
            return 2**31 - 1
        
        return result

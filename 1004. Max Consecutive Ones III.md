# [1004. Max Consecutive Ones III](https://leetcode.com/problems/max-consecutive-ones-iii/description/)
``` py
class Solution(object):
    def longestOnes(self, nums, k):
        l = r = 0
        maxlen = 0
        zeroes = 0

        while r < len(nums):

            # count the zero in the current window
            if nums[r] == 0:
                zeroes += 1

            # if we have more than k zeroes,
            # move the left pointer until 0 reduces
            if zeroes > k:
                if nums[l] == 0:
                    zeroes -= 1
                l += 1

            # current window has at most k zeroes
            if zeroes <= k:
                sublen = r - l + 1
                maxlen = max(maxlen, sublen)

            r += 1

        return maxlen

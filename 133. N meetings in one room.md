# [133. N meetings in one room](https://takeuforward.org/practice/dsa/n-meetings-in-one-room)

class Solution(object):
    def maxMeetings(self, start, end):

        # Store meetings as [start, end]
        meetings = []

        for i in range(len(start)):
            meetings.append([start[i], end[i]])

        # Sort meetings according to ending time
        meetings.sort(key=lambda x: x[1])

        count = 0
        last_end = -1

        # Check each meeting
        for s, e in meetings:

            # Meeting can be selected only if
            # it starts after the previous meeting ends
            if s > last_end:
                count += 1
                last_end = e

        return count

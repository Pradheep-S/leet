# [185. Find Nth root of a number](https://takeuforward.org/practice/dsa/find-nth-root-of-a-number)
```py
class Solution(object):
    def NthRoot(self, N, M):

        low = 1
        high = M

        while low <= high:
            mid = (low + high) // 2

            value = mid ** N

            if value == M:
                return mid

            elif value < M:
                low = mid + 1

            else:
                high = mid - 1

        return -1

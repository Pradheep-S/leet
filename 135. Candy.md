# [135. Candy](https://leetcode.com/problems/candy/description/)
```py
class Solution(object):
    def candy(self, ratings):
        n = len(ratings)
        left = [0] * n
        right = [0] * n
        res = [0] * n
        # do from left to right
        left[0] = 1
        for i in range(1, n):
            if ratings[i] > ratings[i-1]:
                left[i] = left[i-1] + 1
            else:
                left[i] = 1
        # from right to left the same approach
        right[n-1] = 1
        for i in range(n-2, -1, -1):
            if ratings[i] > ratings[i+1]:
                right[i] = right[i+1]+1
            else:
                right[i] = 1
        # the max of left and right
        for i in range(n):
            res[i] = max(left[i], right[i])
        
        return sum(res)

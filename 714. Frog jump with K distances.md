# [714. Frog jump with K distances](https://takeuforward.org/practice/dsa/frog-jump-with-k-distances)
```py
class Solution(object):
    def frogJump(self, heights, k):
        n = len(heights)

        dp = [0] * n

        for i in range(1, n):
            minEnergy = float('inf')

            # Check all possible previous jumps
            for j in range(1, k + 1):
                if i - j >= 0:
                    energy = dp[i - j] + abs(heights[i] - heights[i - j])
                    minEnergy = min(minEnergy, energy)

            dp[i] = minEnergy

        return dp[n - 1]

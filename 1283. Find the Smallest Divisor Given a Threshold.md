# [1283. Find the Smallest Divisor Given a Threshold](https://leetcode.com/problems/find-the-smallest-divisor-given-a-threshold/description/)
```py
class Solution(object):
    def smallestDivisor(self, nums, threshold):

        low = 1
        high = max(nums)
        ans = high

        while low <= high:
            # middle divisor
            mid = (low + high) // 2
            total = 0
            # sum for this divisor
            for num in nums:
                total += (num + mid - 1) // mid # ceiling division

            # If this divisor works, try to find a smaller one
            if total <= threshold:
                ans = mid
                high = mid - 1
            # If it doesn't work, we need a larger divisor
            else:
                low = mid + 1

        return ans

# [1011. Capacity To Ship Packages Within D Days](https://leetcode.com/problems/capacity-to-ship-packages-within-d-days/description/)
```py
class Solution(object):
    def shipWithinDays(self, weights, days):
        def didmove(cap):
            # We start with the first day
            daystaken = 1
            total = 0
            for weight in weights:
                # If package fits in the current ship
                if total + weight <= cap:
                    total += weight
                # or use a new ship/day
                else:
                    daystaken += 1
                    total = weight # 1st pack of next ship is added here
            return daystaken <= days
        # minimum possible = heaviest package
        low = max(weights)
        # maximum possible = total weight
        high = sum(weights)
        res = high
        while low <= high:
            # mid represents the ship capacity
            mid = (low + high) // 2
            if didmove(mid):
                res = mid
                high = mid - 1
            else:
                low = mid + 1

        return res

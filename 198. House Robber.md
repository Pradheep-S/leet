# [198. House Robber](https://leetcode.com/problems/house-robber/description/)
```py
class Solution(object):
    def rob(self, nums):
        n = len(nums)
        if n == 1:
            return nums[0]
        dp = [0] * n
        dp[0] = nums[0]
        dp[1] = max(nums[0], nums[1])

        for i in range(2, n):
            # Don't rob current house
            notRob = dp[i - 1]
            # Rob current house
            rob = dp[i - 2] + nums[i]

            dp[i] = max(notRob, rob)

        return dp[n - 1]
